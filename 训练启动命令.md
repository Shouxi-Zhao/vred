# ---------------------------------------------------------------------------
# 可选 L_ewkd（effect-weighted final-logit KD）
# 不加 --effect-weighted-kd 时保持原 DFT 路径；以下为 K9 + 小权重普通 KD + L_ewkd。
# 比例：lambda_logits=0.2, lambda_effect_kd=1.0（1:5）；lambda_effect_kd=2 不是默认推荐。
# 24G 显存建议 micro=1/accum=16；48G/96G 可用下面 2/8，global batch 仍为 96。
python -m vred.launch dft --gpus 6 --seed 1234 --epochs 1 --lr 2e-5 \
  --micro-batch-size 2 --accumulation 8 --pairs 9 \
  --lambda-write 25 --lambda-logits 0.2 \
  --effect-weighted-kd --lambda-effect-kd 1.0 \
  --num-workers 4 \
  --ce-chunk-size 256 --query-chunk-size 64 --probe-vocab-chunk-size 16384 \
  --checkpoint-probes --selective-checkpointing \
  --input-model checkpoints/vred/sft \
  --output checkpoints/vred/dft-ewkd

# A：hidden-effect 权重 + floor（L_ewkd 使用 teacher 隐藏态 effect，不需要第二次 teacher forward）
# w_t = floor + (1-floor) * s_t / mean(s)，s_t = mean_pair ||u_plus-u_minus||_2
# 权重已归一化到均值 1，因此 lambda_effect_kd 从 0.5 起，不要直接沿用 logits 模式的 1.0。
# 要求：write_position_scope=response、不要传 write_position_limit、write_effect_sampling=uniform。
# 24G 显存若 OOM，先降 query-chunk-size 到 32，再试 micro=1/accum=16。
python -m vred.launch dft --gpus 6 --seed 1234 --epochs 1 --lr 2e-5 \
  --micro-batch-size 2 --accumulation 8 --pairs 9 \
  --lambda-write 25 --lambda-logits 0 \
  --effect-weighted-kd --effect-kd-weight-source hidden \
  --effect-kd-floor 0.15 --lambda-effect-kd 0.5 \
  --write-position-scope response --write-effect-sampling uniform \
  --num-workers 4 \
  --ce-chunk-size 256 --query-chunk-size 64 --probe-vocab-chunk-size 16384 \
  --checkpoint-probes --selective-checkpointing \
  --input-model checkpoints/vred/sft \
  --output checkpoints/vred/dft-ewkd-hidden
